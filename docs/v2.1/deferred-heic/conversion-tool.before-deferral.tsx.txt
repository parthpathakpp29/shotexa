"use client";
/* eslint-disable @next/next/no-img-element -- previews use short-lived local Blob URLs, not remote images. */

import { useEffect, useRef, useState } from "react";
import { Button } from "@/components/ui/button";
import { ContinueWith } from "@/components/workspace/continue-with";
import { useWorkspaceContext } from "@/components/workspace/workspace-provider";
import { continuationsFor } from "@/config/tools";
import { convertHeic } from "@/core/conversion/heic";
import { openPdf, renderPdfPage } from "@/core/conversion/pdf-images";
import { assertRasterSize, baseName, type RasterFormat } from "@/core/conversion/raster";
import { convertSvg, sanitizeSvg } from "@/core/conversion/svg";
import { downloadAsset, downloadBlob } from "@/core/runtime/runtime";

type Kind = "heic-to-jpg" | "pdf-to-jpg" | "svg-to-png";
type Result = { id: string; name: string; width: number; height: number; error?: never } | { id?: never; name: string; error: string; width?: never; height?: never };

function errorText(error: unknown): string { return error instanceof Error ? error.message : "Conversion failed. Try another file."; }

export function ConversionTool({ tool }: { tool: Kind }) {
  const { runtime } = useWorkspaceContext();
  const [format, setFormat] = useState<RasterFormat>(tool === "svg-to-png" ? "png" : "jpg");
  const [quality, setQuality] = useState(92);
  const [busy, setBusy] = useState(false);
  const [progress, setProgress] = useState("");
  const [error, setError] = useState("");
  const [results, setResults] = useState<Result[]>([]);
  const [heicFiles, setHeicFiles] = useState<File[]>([]);
  const [pdfName, setPdfName] = useState("");
  const pdfRef = useRef<Awaited<ReturnType<typeof openPdf>> | null>(null);
  const previewUrls = useRef<string[]>([]);
  const generation = useRef(0);
  const [pageCount, setPageCount] = useState(0);
  const [selectedPages, setSelectedPages] = useState<number[]>([]);
  const [pagePreviews, setPagePreviews] = useState<Record<number, string>>({});
  const [scale, setScale] = useState(1.5);
  const [svg, setSvg] = useState("");
  const [svgName, setSvgName] = useState("");
  const [svgPreview, setSvgPreview] = useState("");
  const [originalSize, setOriginalSize] = useState({ width: 1, height: 1 });
  const [width, setWidth] = useState(1024);
  const [height, setHeight] = useState(1024);
  const [lockRatio, setLockRatio] = useState(true);
  const [background, setBackground] = useState("#ffffff");

  useEffect(() => () => {
    generation.current++;
    previewUrls.current.forEach(URL.revokeObjectURL);
    void pdfRef.current?.destroy();
  }, []);

  const clearPreviews = () => { previewUrls.current.forEach(URL.revokeObjectURL); previewUrls.current = []; setPagePreviews({}); };
  const clearSvgPreview = () => {
    previewUrls.current.forEach(URL.revokeObjectURL);
    previewUrls.current = [];
    setSvgPreview("");
  };
  const addResult = (blob: Blob, name: string, width: number, height: number): Result => {
    const id = runtime.addConvertedImage(blob, name, width, height, tool);
    return { id, name, width, height };
  };

  async function chooseFiles(files: FileList | null) {
    if (!files?.length || busy) return;
    setError(""); setResults([]); setProgress("");
    if (tool === "heic-to-jpg") {
      setHeicFiles([]);
      if (files.length > 20) { setError("Choose up to 20 files at a time."); return; }
      setHeicFiles(Array.from(files));
      return;
    }
    const file = files[0];
    if (tool === "svg-to-png") {
      clearSvgPreview();
      setSvg(""); setSvgName("");
      if (file.size > 2_000_000) { setError("Choose an SVG under 2 MB."); return; }
      try {
        const parsed = sanitizeSvg(await file.text());
        const url = URL.createObjectURL(new Blob([parsed.svg], { type: "image/svg+xml" }));
        previewUrls.current.push(url);
        setSvgPreview(url); setSvg(parsed.svg); setSvgName(file.name);
        setOriginalSize({ width: parsed.width, height: parsed.height });
        setWidth(Math.round(parsed.width)); setHeight(Math.round(parsed.height));
      } catch (e) { setSvg(""); setSvgName(""); setError(errorText(e)); }
      return;
    }
    setBusy(true); setPdfName(""); setPageCount(0); setSelectedPages([]); clearPreviews();
    const thisGeneration = ++generation.current;
    try {
      await pdfRef.current?.destroy(); pdfRef.current = null;
      setProgress("Opening PDF…");
      const doc = await openPdf(file);
      if (generation.current !== thisGeneration) { await doc.destroy(); return; }
      pdfRef.current = doc; setPdfName(file.name); setPageCount(doc.numPages);
      setSelectedPages(Array.from({ length: doc.numPages }, (_, i) => i + 1));
      for (let n = 1; n <= doc.numPages; n++) {
        if (generation.current !== thisGeneration) break;
        setProgress(`Preparing page previews ${n} of ${doc.numPages}…`);
        try {
          const page = await doc.getPage(n);
          const size = page.getViewport({ scale: 1 });
          page.cleanup();
          const previewScale = Math.min(0.22, 240 / Math.max(size.width, size.height));
          const preview = await renderPdfPage(doc, n, previewScale, "png");
          const url = URL.createObjectURL(preview.blob);
          previewUrls.current.push(url);
          setPagePreviews((current) => ({ ...current, [n]: url }));
        } catch { /* The page can still be selected and exported at a safer scale. */ }
        await new Promise((resolve) => setTimeout(resolve, 0));
      }
      setProgress("");
    } catch (e) { setError(errorText(e)); setProgress(""); }
    finally { setBusy(false); }
  }

  async function convert() {
    if (busy) return;
    setBusy(true); setError(""); setResults([]);
    const output: Result[] = [];
    try {
      if (tool === "heic-to-jpg") {
        for (let i = 0; i < heicFiles.length; i++) {
          const file = heicFiles[i]; setProgress(`Converting ${i + 1} of ${heicFiles.length}…`);
          try {
            const converted = await convertHeic(file, format, quality / 100);
            output.push(addResult(converted.blob, `${baseName(file.name)}.${format}`, converted.width, converted.height));
          } catch (e) { output.push({ name: file.name, error: errorText(e) }); }
          setResults([...output]);
          await new Promise((resolve) => setTimeout(resolve, 0));
        }
      } else if (tool === "pdf-to-jpg") {
        const doc = pdfRef.current;
        if (!doc || !selectedPages.length) throw new Error("Select at least one PDF page.");
        let totalPixels = 0;
        for (const n of selectedPages) {
          const page = await doc.getPage(n);
          const viewport = page.getViewport({ scale });
          assertRasterSize(Math.ceil(viewport.width), Math.ceil(viewport.height));
          totalPixels += Math.ceil(viewport.width) * Math.ceil(viewport.height);
          page.cleanup();
          if (totalPixels > 80_000_000) throw new Error("This selection exceeds the safe batch size. Select fewer pages or a lower resolution.");
        }
        for (let i = 0; i < selectedPages.length; i++) {
          const n = selectedPages[i]; setProgress(`Rendering page ${n} (${i + 1} of ${selectedPages.length})…`);
          try {
            const converted = await renderPdfPage(doc, n, scale, format, quality / 100);
            output.push(addResult(converted.blob, `${baseName(pdfName)}-page-${String(n).padStart(2, "0")}.${format}`, converted.width, converted.height));
          } catch (e) { output.push({ name: `Page ${n}`, error: errorText(e) }); }
          setResults([...output]);
          await new Promise((resolve) => setTimeout(resolve, 0));
        }
      } else {
        if (!svg) throw new Error("Choose an SVG first.");
        setProgress("Rasterizing SVG…");
        const blob = await convertSvg(svg, width, height, format, background, quality / 100);
        output.push(addResult(blob, `${baseName(svgName)}.${format}`, width, height));
        setResults([...output]);
      }
      if (output.every((item) => item.error)) setError("No images were converted. See the file errors below.");
    } catch (e) { setError(errorText(e)); }
    finally { setProgress(""); setBusy(false); }
  }

  async function downloadZip() {
    const entries = results.flatMap((item) => {
      if (!item.id) return [];
      const blob = runtime.registry.blob(item.id);
      return blob ? [{ name: item.name, input: blob }] : [];
    });
    if (!entries.length) return;
    try {
      setBusy(true);
      const archive = await (await import("@/core/archive/zip")).createZip(entries);
      downloadBlob(archive.blob, `${tool}-images.zip`);
    } catch (e) { setError(errorText(e)); }
    finally { setBusy(false); }
  }

  const success = results.filter((item) => !!item.id);
  const accept = tool === "heic-to-jpg" ? ".heic,.heif,image/heic,image/heif" : tool === "pdf-to-jpg" ? ".pdf,application/pdf" : ".svg,image/svg+xml";
  const ready = tool === "heic-to-jpg" ? heicFiles.length > 0 : tool === "pdf-to-jpg" ? pageCount > 0 && selectedPages.length > 0 : !!svg;
  const chosen = tool === "heic-to-jpg" ? heicFiles.length === 1 ? heicFiles[0].name : heicFiles.length > 1 ? `${heicFiles.length} files selected` : "No files selected" : tool === "pdf-to-jpg" ? pdfName || "No PDF selected" : svgName || "No SVG selected";
  return <div className="rounded-2xl border border-line bg-surface p-5 shadow-sm sm:p-6" data-testid={`${tool}-converter`}>
    <label className="block cursor-pointer rounded-xl border border-dashed border-line-strong bg-surface-3/60 p-5 text-center hover:border-accent focus-within:border-accent focus-within:ring-2 focus-within:ring-accent/30">
      <span className="block text-base font-semibold text-ink">{tool === "heic-to-jpg" ? "Choose HEIC / HEIF photos" : tool === "pdf-to-jpg" ? "Choose a PDF" : "Choose an SVG"}</span>
      <span className="t-body-sm mt-1 block text-ink-3">Processed in this browser only</span>
      <input type="file" accept={accept} multiple={tool === "heic-to-jpg"} disabled={busy} aria-label={tool === "heic-to-jpg" ? "HEIC or HEIF files" : tool === "pdf-to-jpg" ? "PDF file" : "SVG file"} onChange={(event) => { void chooseFiles(event.target.files); event.target.value = ""; }} className="sr-only" />
      <span className="mt-4 flex items-center justify-center gap-3 text-sm"><span className="rounded-md bg-accent px-4 py-2 font-semibold text-white">Choose {tool === "heic-to-jpg" ? "files" : "file"}</span><span className="max-w-48 truncate text-ink-2" title={chosen}>{chosen}</span></span>
    </label>
    {tool === "heic-to-jpg" && heicFiles.length > 0 && <p className="t-body-sm mt-3 text-ink-2">{heicFiles.length} file{heicFiles.length === 1 ? "" : "s"} selected · maximum 25 MB each</p>}
    {tool === "pdf-to-jpg" && pageCount > 0 && <>
      <div className="mt-5 flex flex-wrap items-center justify-between gap-3"><p className="font-semibold">{pdfName} · {pageCount} pages</p><div className="flex gap-2"><Button size="sm" variant="secondary" onClick={() => setSelectedPages(Array.from({ length: pageCount }, (_, i) => i + 1))}>Select all</Button><Button size="sm" variant="secondary" onClick={() => setSelectedPages([])}>Clear</Button></div></div>
      <div className="mt-3 grid max-h-80 grid-cols-3 gap-2 overflow-y-auto sm:grid-cols-4" aria-label="PDF pages">
        {Array.from({ length: pageCount }, (_, i) => i + 1).map((n) => <label key={n} className="cursor-pointer rounded-md border border-line p-2 text-center text-sm hover:border-accent"><input type="checkbox" checked={selectedPages.includes(n)} onChange={() => setSelectedPages((pages) => pages.includes(n) ? pages.filter((p) => p !== n) : [...pages, n].sort((a,b) => a-b))} aria-label={`Page ${n}`} className="mr-1" /><span>Page {n}</span>{pagePreviews[n] && <img src={pagePreviews[n]} alt={`Preview of PDF page ${n}`} className="mt-2 h-24 w-full object-contain" />}</label>)}
      </div>
      <label className="mt-4 block text-sm font-medium">Resolution <select value={scale} onChange={(e) => setScale(Number(e.target.value))} className="ml-2 rounded-md border border-line bg-surface px-3 py-2"><option value={1}>Standard · 1×</option><option value={1.5}>Sharp · 1.5×</option><option value={2}>High · 2×</option><option value={3}>Very high · 3×</option></select></label>
    </>}
    {tool === "svg-to-png" && svg && <>
      <div className="mt-4 flex min-h-40 items-center justify-center rounded-lg border border-line bg-[linear-gradient(45deg,#eee_25%,transparent_25%),linear-gradient(-45deg,#eee_25%,transparent_25%),linear-gradient(45deg,transparent_75%,#eee_75%),linear-gradient(-45deg,transparent_75%,#eee_75%)] bg-[length:20px_20px]">{svgPreview && <img src={svgPreview} alt="Sanitized SVG preview" className="max-h-48 max-w-full object-contain" />}</div>
      <p className="t-body-sm mt-2 text-ink-2">{svgName} · source {originalSize.width} × {originalSize.height}</p>
      <div className="mt-4 grid grid-cols-2 gap-3"><label className="text-sm font-medium">Width <input type="number" min={1} max={16384} value={width} onChange={(e) => { const next = Number(e.target.value); setWidth(next); if (lockRatio) setHeight(Math.max(1, Math.round(next * originalSize.height / originalSize.width))); }} className="mt-1 w-full rounded-md border border-line bg-surface px-3 py-2" /></label><label className="text-sm font-medium">Height <input type="number" min={1} max={16384} value={height} onChange={(e) => { const next = Number(e.target.value); setHeight(next); if (lockRatio) setWidth(Math.max(1, Math.round(next * originalSize.width / originalSize.height))); }} className="mt-1 w-full rounded-md border border-line bg-surface px-3 py-2" /></label></div>
      <label className="mt-2 flex items-center gap-2 text-sm"><input type="checkbox" checked={lockRatio} onChange={(e) => setLockRatio(e.target.checked)} /> Preserve aspect ratio</label>
      {format === "jpg" && <label className="mt-3 flex items-center gap-2 text-sm">JPG background <input type="color" value={background} onChange={(e) => setBackground(e.target.value)} aria-label="JPG background color" /></label>}
    </>}
    <div className="mt-5 flex flex-wrap items-end gap-4">
      <label className="text-sm font-medium">Output format <select value={format} onChange={(e) => setFormat(e.target.value as RasterFormat)} className="mt-1 block min-h-11 rounded-md border border-line bg-surface px-3"><option value="jpg">JPG</option><option value="png">PNG</option></select></label>
      {format === "jpg" && <label className="min-w-36 flex-1 text-sm font-medium">JPG quality: {quality}% <input type="range" min={60} max={100} value={quality} onChange={(e) => setQuality(Number(e.target.value))} className="mt-3 block w-full" /></label>}
      <Button variant="primary" size="lg" disabled={!ready || busy} onClick={() => void convert()}>{busy ? "Working…" : "Convert"}</Button>
    </div>
    {(progress || error) && <p role={error ? "alert" : "status"} className={`mt-4 text-sm ${error ? "text-red-700" : "text-ink-2"}`}>{error || progress}</p>}
    {results.length > 0 && <section className="mt-6 border-t border-line pt-5" aria-label="Conversion results"><div className="flex flex-wrap items-center justify-between gap-3"><h2 className="t-h3">Results ({success.length}/{results.length})</h2>{success.length > 1 && <Button variant="secondary" disabled={busy} onClick={() => void downloadZip()}>Download ZIP</Button>}</div><ul className="mt-3 space-y-2">{results.map((item, index) => <li key={`${item.name}-${index}`} className="flex items-center gap-3 rounded-md border border-line bg-surface-3 p-3">{item.id && <img src={runtime.registry.objectUrl(item.id)} alt={`Converted preview of ${item.name}`} loading="lazy" className="h-12 w-16 shrink-0 rounded-sm border border-line bg-white object-contain" />}<div className="min-w-0 flex-1"><p className="truncate text-sm font-medium">{item.name}</p><p className="text-xs text-ink-3">{item.error ?? `${item.width} × ${item.height} px · Added to workspace`}</p></div>{item.id && <Button size="sm" variant="secondary" onClick={() => downloadAsset(runtime, item.id!)}>Download</Button>}</li>)}</ul>{success.length > 0 && <ContinueWith tools={continuationsFor(tool)} fileId={success[0].id} />}</section>}
  </div>;
}
